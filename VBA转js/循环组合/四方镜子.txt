Function zxgbs(hs As Boolean)
    Dim 最右列 As Long, i, ii, iii, Arr, brr, crr, drr, 最小公倍数, mm, 源, 积, 相等
    最右列 = Cells(1, Columns.count).End(xlToLeft).Column
    ReDim Arr(1 To 最右列)
    积 = 1: 相等 = False
    For i = 1 To UBound(Arr)
        Arr(i) = Cells(rows.count, i).End(xlUp).row
        积 = 积 * Arr(i)
    Next
    最小公倍数 = Application.WorksheetFunction.Lcm(Arr)
    If 最小公倍数 > 1048576 Then
        MsgBox "已超出表格限制"
        Exit Function
    End If
    If 积 - 最小公倍数 = 0 Then
        相等 = True
    End If
    源 = Range(Cells(1, 1), Cells(最小公倍数, 最右列))
    Sheets.Add After:=Sheets("重复字")
    If 相等 Then
        ActiveSheet.Name = "完整_" & 最小公倍数 & "sheet" & Sheets.count
    Else
        ActiveSheet.Name = "残缺_" & 积 & "|" & 最小公倍数 & "sheet" & Sheets.count
    End If
    ReDim crr(1 To 最小公倍数)
    If hs Then
        ReDim brr(1 To 最小公倍数, 1 To 最右列)
        ReDim drr(1 To 最小公倍数, 1 To 1)
        For ii = 1 To 最小公倍数
            mm = ""
            For iii = 1 To 最右列
                brr(ii, iii) = 源(ys(ii, Arr(iii)), iii)
                mm = mm & brr(ii, iii) & mmm
            Next
            drr(ii, 1) = mm
        Next
        If CheckBox1.Value Then
            [a1].Resize(UBound(brr), 1) = drr
        Else
            [a1].Resize(UBound(brr), UBound(brr, 2)) = brr
        End If
    Else
        ReDim brr(1 To 最右列, 1 To 最小公倍数)
        For ii = 1 To 最小公倍数
            mm = ""
            For iii = 1 To 最右列
                brr(iii, ii) = 源(ys(ii, Arr(iii)), iii)
                mm = mm & brr(iii, ii) & mmm
            Next
            crr(ii) = mm
        Next
        If CheckBox1.Value Then
            [a1].Resize(1, UBound(brr, 2)) = crr
        Else
            [a1].Resize(UBound(brr), UBound(brr, 2)) = brr
        End If
    End If
End Function

Function scct(js As Boolean, hs As Boolean)
    On Error Resume Next
    Dim n, nn, m, mm, L, LL, lll, LLLL
    m = Range("xfd1").End(xlToLeft).Column
    ReDim L(1 To 4, 1 To m)
    L(2, 1) = 1
    For n = 1 To m
        L(1, n) = Cells(1048576, n).End(xlUp).row
        L(2, n) = L(2, n - 1) * L(1, n - 1)
        L(3, n) = L(1, n) * L(2, n)
    Next
    For n = 1 To m
        L(4, n) = L(3, m) / L(3, n)
    Next
    If Application.WorksheetFunction.max(L) > 1048576 Then
        MsgBox "已超出表格限制"
        Exit Function
    End If
    LL = Range(Cells(1, 1), Cells(L(3, m), m))
    Select Case True
        Case js And hs
            ReDim lll(1 To m, 1 To L(3, m))
            For nn = 1 To m
                For n = 1 To L(3, m)
                    lll(nn, n) = LL(ys(cd(n, L(2, nn)), (L(1, nn))), nn)
                Next
            Next
            If CheckBox1.Value Then
                ReDim LLLL(1 To L(3, m))
                For nn = 1 To L(3, m)
                    mm = ""
                    For n = 1 To m
                        mm = mm & lll(n, nn) & mmm
                    Next
                    LLLL(nn) = mm
                Next
                [F2].Resize(1, L(3, m)) = LLLL
            Else
                [F2].Resize(UBound(lll), UBound(lll, 2)) = lll
            End If
        Case js And Not hs
            ReDim lll(1 To L(3, m), 1 To m)
            For nn = 1 To m
                For n = 1 To L(3, m)
                    lll(n, nn) = LL(ys(cd(n, L(2, nn)), (L(1, nn))), nn)
                Next
            Next
            If CheckBox1.Value Then
                ReDim LLLL(1 To L(3, m), 1 To 1)
                For n = 1 To L(3, m)
                    mm = ""
                    For nn = 1 To m
                        mm = mm & lll(n, nn) & mmm
                    Next
                    LLLL(n, 1) = mm
                Next
                [F2].Resize(UBound(LLLL), UBound(LLLL, 2)) = LLLL
            Else
                [F2].Resize(UBound(lll), UBound(lll, 2)) = lll
            End If
        Case Not js And hs
            ReDim lll(1 To m, 1 To L(3, m))
            For nn = 1 To m
                For n = 1 To L(3, m)
                    lll(nn, n) = LL(ys(cd(n, L(4, nn)), (L(1, nn))), nn)
                Next
            Next
            If CheckBox1.Value Then
                ReDim LLLL(1 To L(3, m))
                For nn = 1 To L(3, m)
                    mm = ""
                    For n = 1 To m
                        mm = mm & lll(n, nn) & mmm
                    Next
                    LLLL(nn) = mm
                Next
                [F2].Resize(1, L(3, m)) = LLLL
            Else
                [F2].Resize(UBound(lll), UBound(lll, 2)) = lll
            End If
        Case Not js And Not hs
            ReDim lll(1 To L(3, m), 1 To m)
            For nn = 1 To m
                For n = 1 To L(3, m)
                    lll(n, nn) = LL(ys(cd(n, L(4, nn)), (L(1, nn))), nn)
                Next
            Next
            If CheckBox1.Value Then
                ReDim LLLL(1 To L(3, m), 1 To 1)
                For n = 1 To L(3, m)
                    mm = ""
                    For nn = 1 To m
                        mm = mm & lll(n, nn) & mmm
                    Next
                    LLLL(n, 1) = mm
                Next
                [F2].Resize(UBound(LLLL), UBound(LLLL, 2)) = LLLL
            Else
                [F2].Resize(UBound(lll), UBound(lll, 2)) = lll
            End If
        Case Else
            MsgBox "未知错误"
    End Select
End Function

Function ys(n, Y)
    ys = ((n + Y - 1) Mod Y) + 1
End Function

Function cd(c, d)
    cd = Application.WorksheetFunction.RoundUp(c / d, 0)
End Function
